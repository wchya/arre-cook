func TestCRCloneBypassesQuota(t *testing.T) {
 u,tok,err:=testutil.NewUser("cr-clone@example.test"); if err!=nil {t.Fatal(err)}
 rows:=make([]models.Dish,500)
 for i:=range rows {rows[i]=models.Dish{OwnerID:u.ID,Name:fmt.Sprintf("quota-%d",i),MealType:"all",Enabled:true,Ingredients:"[]",Steps:"[]"}}
 if err:=database.DB.CreateInBatches(&rows,50).Error;err!=nil {t.Fatal(err)}
 st,r:=call(t,"POST","/api/dishes",tok,map[string]any{"name":"normal-create"});must(t,st,r,400)
 st,r=call(t,"POST",fmt.Sprintf("/api/dishes/%d/clone",rows[0].ID),tok,nil);must(t,st,r,200)
 var n int64;database.DB.Model(&models.Dish{}).Where("owner_id = ? AND family_id = 0",u.ID).Count(&n)
 if n!=501 {t.Fatalf("count=%d",n)}
 t.Logf("normal creation blocked; clone succeeded; private recipe count=%d",n)
}

func TestCRFailedSecurityUpdatesReportSuccess(t *testing.T) {
 u,tok,err:=testutil.NewUser("cr-update@example.test");if err!=nil {t.Fatal(err)}
 name:="cr_users_update_failure"
 database.DB.Callback().Update().Before("gorm:update").Register(name,func(tx *gorm.DB){if tx.Statement.Table=="users" {tx.AddError(errors.New("CR injected write failure"))}})
 defer database.DB.Callback().Update().Remove(name)
 st,r:=call(t,"PUT","/api/me/password",tok,map[string]string{"new_password":"ValidNewPass123!"});must(t,st,r,200)
 st,r=call(t,"POST","/api/me/logout-all",tok,nil);must(t,st,r,200)
 var fresh models.User;if err:=database.DB.First(&fresh,u.ID).Error;err!=nil {t.Fatal(err)}
 if fresh.TokenVersion!=u.TokenVersion||fresh.PasswordHash!="" {t.Fatal("injected failure did not preserve DB state")}
 st,r=call(t,"GET","/api/me",tok,nil);must(t,st,r,200)
 t.Log("password change and logout-all returned HTTP 200; password/version unchanged; original token still works")
}
func TestCRMenuCacheSurvivesDishEdit(t *testing.T) {
 u,tok,err:=testutil.NewUser("cr-cache@example.test");if err!=nil {t.Fatal(err)}
 d:=models.Dish{OwnerID:u.ID,Name:"old name",Enabled:true,MealType:"all",Ingredients:`[{"name":"old ingredient","amount":"1g"}]`}
 if err:=database.DB.Create(&d).Error;err!=nil {t.Fatal(err)}
 now:=time.Now();weekday:=int(now.Weekday());if weekday==0{weekday=7};monday:=now.AddDate(0,0,1-weekday).Format("2006-01-02")
 b,_:=json.Marshal(services.WeekPlan{Days:[]services.WeekDayPlan{{Date:monday,Lunch:[]models.Dish{d},Dinner:[]models.Dish{}}}})
 if err:=database.SetUserSetting(u.ID,"week_plan_cache",string(b));err!=nil{t.Fatal(err)}

func TestCRTransferFamilyCanRaceWithLeave(t *testing.T) {
 owner,_,err:=testutil.NewUser("cr-owner@example.test");if err!=nil {t.Fatal(err)}
 member,_,err:=testutil.NewUser("cr-member@example.test");if err!=nil {t.Fatal(err)}
 family,err:=services.CreateFamily(owner.ID,"CR family");if err!=nil {t.Fatal(err)}
 if err:=database.DB.Create(&models.FamilyMember{FamilyID:family.ID,UserID:member.ID,JoinedAt:time.Now()}).Error;err!=nil {t.Fatal(err)}
 name:="cr_family_interleaving";ran:=false
 database.DB.Callback().Update().Before("gorm:begin_transaction").Register(name,func(tx *gorm.DB){
  if tx.Statement.Table=="families"&&!ran {ran=true;if err:=services.LeaveFamily(member.ID);err!=nil {tx.AddError(err)}}
 })
 defer database.DB.Callback().Update().Remove(name)
 if err:=services.TransferFamily(owner.ID,member.ID);err!=nil {t.Fatal(err)}
 var fresh models.Family;database.DB.First(&fresh,family.ID)
 var n int64;database.DB.Model(&models.FamilyMember{}).Where("family_id = ? AND user_id = ?",family.ID,member.ID).Count(&n)
 if !ran||fresh.OwnerID!=member.ID||n!=0 {t.Fatalf("ran=%v owner=%d membership=%d",ran,fresh.OwnerID,n)}
 if err:=services.RenameFamily(owner.ID,"owner tries");err==nil {t.Fatal("old owner still manages")}
 if err:=services.RenameFamily(member.ID,"new owner tries");err==nil {t.Fatal("departed new owner still manages")}
 t.Log("member left between transfer validation and UPDATE; transfer succeeded; neither original nor new owner can manage the family")
}
func TestCRPasswordWriteCanRestoreRevokedGeneration(t *testing.T) {
 u,tok,err:=testutil.NewUser("cr-generation@example.test");if err!=nil {t.Fatal(err)}
 previous:=u;previous.TokenVersion=2
 revokedToken,_,err:=auth.IssueUserToken(&previous);if err!=nil {t.Fatal(err)}
 name:="cr_version_interleaving";ran:=false
 database.DB.Callback().Update().Before("gorm:begin_transaction").Register(name,func(tx *gorm.DB){
  if tx.Statement.Table=="users"&&!ran {
   ran=true
   // Two already-committed revocations advance the generation after this handler read it.
   if err:=database.DB.Model(&models.User{}).Where("id = ?",u.ID).UpdateColumn("token_version",gorm.Expr("token_version + 2")).Error;err!=nil{tx.AddError(err);return}
   st,r:=call(t,"GET","/api/me",revokedToken,nil);must(t,st,r,401)
  }
 })
 defer database.DB.Callback().Update().Remove(name)
 st,r:=call(t,"PUT","/api/me/password",tok,map[string]string{"new_password":"ValidNewPass123!"});must(t,st,r,200)
 var fresh models.User;database.DB.First(&fresh,u.ID)
 if !ran||fresh.TokenVersion!=2 {t.Fatalf("ran=%v version=%d",ran,fresh.TokenVersion)}
 st,r=call(t,"GET","/api/me",revokedToken,nil);must(t,st,r,200)
 t.Log("delayed password write rolled token version 3 back to 2; token rejected before write was accepted afterwards")
}
